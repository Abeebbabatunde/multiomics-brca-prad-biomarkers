> ## ---- 2. Detect ID type in a vector of feature IDs ---------------------------
> 
> detect_id_type <- function(ids) {
+   ids <- na.omit(ids)
+   if (length(ids) == 0) return("EMPTY")
+   n <- length(ids)
+   n_cpg     <- sum(str_detect(ids, "^cg[0-9]+$"))
+   n_ensembl <- sum(str_detect(ids, "^ENS[GTP][0-9]+"))
+   n_uniprot <- sum(str_detect(ids, "^[A-NR-Z][0-9][A-Z0-9]{3}[0-9]$|^[OPQ][0-9][A-Z0-9]{3}[0-9]$"))
+   n_entrez  <- sum(str_detect(ids, "^[0-9]+$"))
+   n_symbol  <- sum(str_detect(ids, "^[A-Za-z][A-Za-z0-9\\-\\.]*$")) - n_entrez
+   
+   props <- c(CpG_probe = n_cpg, Ensembl = n_ensembl, UniProt = n_uniprot,
+              Entrez = n_entrez, Gene_symbol_likely = n_symbol) / n
+   best <- names(which.max(props))
+   sprintf("%s (%.0f%% match, n=%d) | example IDs: %s",
+           best, props[best] * 100, n, paste(head(ids, 3), collapse = ", "))
+ }

+     add("MOFA file: %s", mf)
+     obj <- tryCatch({

+         e <- new.env(); load(mf, envir = e)
+         mofa_candidates <- Filter(function(n) inherits(get(n, envir = e), "MOFA"), ls(e))
