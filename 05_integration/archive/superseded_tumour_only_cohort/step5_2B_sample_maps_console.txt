> tcga_patient_id <- function(x) {
+   
+   x <- as.character(x)
+   
+   # Convert common separators to hyphens
+   x <- gsub("\\.", "-", x)
+   x <- gsub("_", "-", x)
+   
+   # Extract TCGA patient barcode
+   out <- sub(
+     "^(TCGA-[A-Za-z0-9]+-[A-Za-z0-9]+).*",
+     "\\1",
+     x
+   )
+   
+   # Keep only valid TCGA patient IDs
+   out[!grepl(
+     "^TCGA-[A-Za-z0-9]+-[A-Za-z0-9]+$",
+     out
+   )] <- NA
+   
+   out
+ }
