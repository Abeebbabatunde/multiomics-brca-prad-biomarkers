> tcga_sample_code <- function(x) {
+   
+   x <- as.character(x)
+   
+   out <- sub(
+     "^TCGA-[^-]+-[^-]+-([0-9]{2}).*",
+     "\\1",
+     x
+   )
+   
+   out[
+     !grepl(
+       "^TCGA-[^-]+-[^-]+-[0-9]{2}",
+       x
+     )
+   ] <- NA
+   
+   out
+ }

> tcga_vial <- function(x) {
+   
+   x <- as.character(x)
+   
+   out <- sub(
+     "^TCGA-[^-]+-[^-]+-[0-9]{2}([A-Z]).*",
+     "\\1",
+     x
+   )
+   
+   out[
+     !grepl(
+       "^TCGA-[^-]+-[^-]+-[0-9]{2}[A-Z]",
+       x
+     )

+
